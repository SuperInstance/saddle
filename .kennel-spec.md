## Mapping (Casey's seed → schema; every clause is a spec line)

| kennel | RL/alignment | schema element |
|---|---|---|
| treats | reward signal | `RewardSpec` in each curriculum phase |
| whistles | trigger conditioning | `WhistleSpec` (cue → expects) |
| peer pressure from aligned dogs | few-shot demos from top-aligned peers | `PeerDemonstrationSet` selected from ledger |
| breeding to similar ranch | domain-similar fine-tuning | `CandidateSelection.domainSimilarity` (Jaccard over domain tags) |
| land/family/livestock dynamic & temporal | continual adaptation | `DomainProfile.temporal: true` flag + doc note: ranches drift, re-score regularly |
| younger dogs ready sooner | curriculum compression | phases have shrinking `nudgeBudget` |
| less and less nudging | scaffolding decay / harness thinning | `GraduationCriteria.nudgeRateCeiling` |
| reward internal to kennel psychology | intrinsic reward / internalized norms | graduation = harness layers may THIN (never silently removed) |

// --- ranch & candidate: breeding-to-similar-ranch = domain-similar selection
export interface DomainProfile {
  domainId: string;          // 'ranch.support-triage'
  tags: string[];            // what the land+family+livestock are like
  /** dynamic & temporal: the niche moves — profiles must be re-scored, never assumed static */
  temporal: boolean;
}
export interface CandidateSelection {
  candidateId: string;
  lineageId?: string;        // bloodline (base model family)
  domainTags: string[];      // what this candidate is bred toward
  domainSimilarity: number;  // 0..1 score vs the target ranch
  dispositionNotes?: string; // what the candidate arrives already wanting
}
export function domainSimilarityScore(ranch: DomainProfile, candidateDomainTags: string[]): number;
// Jaccard over tag sets, lowercased+trimmed; identical spelling counts once. Pure.
export function rankCandidates(ranch: DomainProfile, candidates: CandidateSelection[], minSimilarity: number): CandidateSelection[];
// re-scores each candidate (trusting caller-declared tags), filters >= minSimilarity, sorts desc, stable on ties by candidateId.

// --- curriculum: whistles & treats with a schedule
export type RewardSchedule = 'continuous' | 'intermittent'; // every response vs variable-ratio (resists extinction)
export interface WhistleSpec { id: string; cue: string; expects: string; }
export interface RewardSpec {
  id: string;
  on: string;               // condition description ('verdict == worked')
  treat: string;            // what the treat IS ('affirmation: good hold')
  schedule: RewardSchedule;
  weight?: number;          // 0..1, default 1
}
export interface CurriculumPhase {
  phaseId: string;
  order: number;            // strictly increasing from 1
  focus: string;
  whistles: WhistleSpec[];
  rewards: RewardSpec[];
  nudgeBudget: number;      // max human nudges allowed this phase; SHOULD decline phase-over-phase (validated)
  exitCriteria: { minKeepRatio: number; minAttempts: number }; // advance when both clear
}
export interface Curriculum { phases: CurriculumPhase[]; }

// --- peer demonstration: few-shot set from top-aligned ledger entries
export interface PeerDemonstration {
  seq: number;              // ledger seq of the demonstrated entry
  alignmentId: string;      // the demonstrating dog
  cellId: string;
  keepRatio: number;        // that alignment's keep ratio at selection time (snapshot)
}
export interface PeerDemonstrationSet {
  source: 'ledger';         // honest label — only source saddle knows
  minKeepRatio: number;     // only top-aligned dogs demonstrate (>= this)
  maxExamples: number;
  demonstrations: PeerDemonstration[];
}
export interface PeerSelectionOptions {
  minKeepRatio: number;     // recommend >= EARNED_KEEP_THRESHOLD (0.75)
  maxExamples: number;      // 1..200
  cellId?: string;          // restrict demonstrations to one cell's yard
}
export async function selectPeerDemonstrations(ledgerPath: string, opts: PeerSelectionOptions): Promise<PeerDemonstrationSet>;
// Streams the ledger ONCE. Keeps per-alignment worked/failed counters (O(#alignments)).
// A demonstration = an entry with verdict 'worked' whose alignment's FINAL keepRatio >= minKeepRatio
// and total entries >= 5 (no puppy classes). Take newest-first up to maxExamples.
// Snapshot the alignment's final keepRatio into each demonstration. Never loads whole corpus.

// --- graduation: nudge-rate below ceiling = harness may thin
export interface GraduationCriteria {
  nudgeRateCeiling: number;   // 0..1 exclusive bounds (0,1): nudges per attempt
  windowSize: number;         // >= 5 attempts per window
  minAttempts: number;        // >= windowSize, statistical floor before any claim
  consecutiveWindows: number; // >= 1 full windows in a row under the ceiling
  thinningOrder: string[];    // harness layers that may come off, in order ('output-filters' → 'rewards' → …)
}
export interface GraduationDecision {
  graduated: boolean;
  nudgeRate: number;          // over the most recent full window (0 if no full window)
  windowsChecked: number;     // full windows evaluated
  windowsBelowCeiling: number;// most recent CONSECUTIVE windows under ceiling (stops counting at first miss going backward)
  reason: string;             // human-readable, honest (insufficient data / stable under ceiling / still needs the harness)
  // data only — saddle never removes harness layers; the cowboy thins
}
export function evaluateGraduation(nudges: ReadonlyArray<{ attempt: number; nudged: boolean }>, criteria: GraduationCriteria): GraduationDecision;
// nudges: chronological samples (attempt index may have gaps; treat array order as time).
// Full windows = the LAST K windows of windowSize samples (drop leading partial).
// nudgeRate over the most recent FULL window. windowsBelowCeiling counts consecutive full windows,
// newest backward, whose rate <= ceiling. graduated = attempts >= minAttempts AND windowsBelowCeiling >= consecutiveWindows.
// Insufficient data (no full window or attempts < minAttempts) → graduated:false with honest reason. Never throws on empty.

// --- the whole kennel program for one candidate
export type KennelJobStatus = 'draft' | 'approved' | 'graduated'; // honest labels; graduation is CLAIMED by nightcycle data, APPROVED by the cowboy
export interface KennelTrainingJob {
  jobId: string;
  createdAt: string;          // ISO
  ranch: DomainProfile;
  candidate: CandidateSelection;
  curriculum: Curriculum;
  peers: PeerDemonstrationSet;
  graduation: GraduationCriteria;
  status: KennelJobStatus;
  notes?: string;
}
export interface KennelJobCheck { ok: boolean; errors: string[]; }
export function validateKennelJob(job: KennelTrainingJob): KennelJobCheck;

## validateKennelJob rules (each violation → one error string, collect ALL)
- ids: jobId non-empty; ranch.domainId non-empty; candidate.candidateId non-empty.
- ranch.tags: non-empty, no duplicates after lowercase+trim.
- candidate.domainSimilarity in [0,1]; domainTags non-empty.
- curriculum.phases: non-empty; order strictly increasing (does NOT have to start at 1, but must be 1..n sequential? — spec: exactly 1..n in order as given); within a phase: whistle ids unique, reward ids unique, reward weight (if set) in (0,1], schedule in vocabulary; nudgeBudget integer >= 0 and NON-INCREASING across phases (younger dogs, less nudging — a rising budget is an error with the phrase 'nudge budget must not rise'); exitCriteria: minKeepRatio in [0,1], minAttempts >= 1.
- peers: source === 'ledger'; minKeepRatio in [0,1]; maxExamples in [1,200]; every demonstration.keepRatio >= minKeepRatio (a demo from a below-bar dog is a lying label — error); demonstration seqs unique.
- graduation: nudgeRateCeiling in (0,1) exclusive; windowSize >= 5; minAttempts >= windowSize; consecutiveWindows >= 1; thinningOrder non-empty, no duplicate layers.
- status in vocabulary; createdAt parses as a Date (Number.isNaN(Date.parse()) check).
